    const x = ((clientX - wrapRect.left - panX) / zoom) * SS;
    const y = ((clientY - wrapRect.top - panY) / zoom) * SS;
    return {x, y};
  }

  // pointermove events fire based on screen-space movement, not canvas-space
  // movement. Zoomed out, the same hand motion covers much more canvas
  // distance per real sample (screenToCanvas divides by zoom), so any fixed
  // amount of device-pixel noise gets amplified by 1/zoom once it lands in
  // canvas space. Ported from draw.html, which doesn't show this wobble:
  // scale smoothing/deadzone by this factor (capped so it doesn't run away
  // at extreme zoom-out) to keep the *screen-space* noise budget constant.
  function zoomSmoothingFactor(){
    return Math.min(4, Math.max(1, 1 / zoom));
  }

  function getPressure(e){
    // A hovering/contact-threshold pen legitimately reports zero. Treating
    // zero as a missing value made the lightest touch jump to full pressure.
    if(e.pointerType === 'pen' && typeof e.pressure === 'number')
      return Math.max(0, Math.min(1, e.pressure));
    if(e.pointerType === 'mouse') return 1;
    return typeof e.pressure === 'number' ? Math.max(0, Math.min(1, e.pressure)) : 1;
  }
  // Pressure controls diameter only. A continuous mild ease-in works with
  // subpixel area coverage to transition smoothly from faint to fully dark.
  function pressureInfluence(pressure){
    const p = Math.max(0, Math.min(1, pressure == null ? 1 : pressure));
    // A mild ease-in leaves room for a faint -> gray -> dark transition.
    // Shader area correction handles the first near-zero footprint without
    // requiring a discontinuous pressure cutoff.
    return Math.pow(p, 1.2);
  }
  // Let pressure approach a true zero-relative diameter. Only the tiny
  // absolute raster floor below prevents the dab from disappearing entirely.
  const MIN_SIZE_PERCENT = 0;
  // A fifth of a backing pixel stays pale through area-based coverage while
  // remaining consistently visible in the continuous analytic stroke.
  const MIN_ABS_RADIUS_BACKING = 0.05 * SS;
  function pressureRadius(size, pressure){
    // At a nominal 1px brush, the ordinary radius produces a fourfold
    // jump from the smallest representable dab to full pressure. Keep this
    // smallest raster preset within a tighter TVPaint-like subpixel range.
    // Brushes 2px and larger retain their existing full-pressure diameter.
    const onePixelScale = size <= 1 ? 0.5 : 1;
    const full = (size * SS) / 2 * onePixelScale;
    const influence = pressureInfluence(pressure);
    const minR = full * MIN_SIZE_PERCENT;
    const r = minR + (full - minR) * influence;
    return Math.max(MIN_ABS_RADIUS_BACKING, r);
  }

  // Hard Round pressure changes diameter, not paint opacity. The dark center
  // remains at the opacity-slider value even at the lightest pressure; only
  // fractional raster coverage makes the subpixel edge appear gray.
  function pressureAlpha(_pressure){
    return opacity;
  }

    // ---------- GPU resources ----------
    // Every stamp (round dab OR the straight capsule between two dabs) is
    // sent down as a plain bounding-box quad carrying the TRUE shape
    // parameters (p0, p1, r) as per-vertex attributes. The fragment shader
    // computes the exact signed distance from that fragment to the real
    // capsule (a dab is just a zero-length capsule, p0 == p1) and turns it
    // into coverage with a fwidth()-sized step.
    //
    // This replaces the old approach of manually extruding an outline
    // polygon by a fixed feather width and linearly interpolating a
    // "coverage" value across it. That worked for a single isolated shape,
    // but a stroke is hundreds of overlapping dabs+capsules stamped one on
    // top of the other, each contributing its own ~1px feather ring; where
    // those rings stacked along the curve of the stroke the blended result
    // read as a much wider, mushy gradient than any single shape's feather
    // — the "way too blurry" edge. Computing distance analytically per
    // fragment means every dab's edge is exactly as sharp as every other
    // dab's, and it can't compound: a fragment is only ever affected by the
    // ~1-pixel band around its own true boundary, regardless of how many
    // other dabs happen to overlap nearby.
    const strokeShader = device.createShaderModule({ code: `
      struct Uniforms {
        size: vec2f,
        color: vec3f,
        mode: f32,
      };
      @group(0) @binding(0) var<uniform> u: Uniforms;

      struct VSOut {
        @builtin(position) pos: vec4f,
        @location(0) p0: vec2f,
        @location(1) p1: vec2f,
        @location(2) r0: f32,
        @location(3) r1: f32,
        @location(4) alpha: f32,
      };

      @vertex
      fn vs(
        @location(0) position: vec2f,
        @location(1) p0: vec2f,
        @location(2) p1: vec2f,
        @location(3) r0: f32,
        @location(4) r1: f32,
        @location(5) alpha: f32
      ) -> VSOut {
        var out: VSOut;
        let ndc = vec2f(
          (position.x / u.size.x) * 2.0 - 1.0,
          1.0 - (position.y / u.size.y) * 2.0
        );
        out.pos = vec4f(ndc, 0.0, 1.0);
        out.p0 = p0;
        out.p1 = p1;
        out.r0 = r0;
        out.r1 = r1;
        out.alpha = alpha;
        return out;
      }

      // Distance from p to the segment a-b (0 when a == b, i.e. a round dab).
      fn sdSegment(p: vec2f, a: vec2f, b: vec2f) -> f32 {
        let pa = p - a;
        let ba = b - a;
        let denom = dot(ba, ba);
        let h = select(clamp(dot(pa, ba) / denom, 0.0, 1.0), 0.0, denom < 1e-6);
        return length(pa - ba * h);
      }

      @fragment
      fn fs(in: VSOut) -> @location(0) vec4f {
        let eraseColor = vec3f(1.0, 1.0, 1.0);
        let rgb = select(u.color, eraseColor, u.mode > 0.5);
        // @builtin(position).xy is the fragment's exact backing-store pixel
        // coordinate, so we can measure real distance-to-shape right here.
        let pa = in.pos.xy - in.p0;
        let ba = in.p1 - in.p0;
        let denom = dot(ba, ba);
        let rawH = dot(pa, ba) / max(denom, 1e-6);
        let h = clamp(rawH, 0.0, 1.0);
        let localRadius = mix(in.r0, in.r1, h);
        let isRoundDab = denom < 1e-6;
        let roundDabDistance = length(pa) - in.r0;
        // Moving curve pieces use rounded endpoint coverage so consecutive
        // pieces form a true union at turns instead of overlapping flat-end
        // wedges that radiate outward into bow-tie/fan artifacts.
        let segmentDistance = length(pa - ba * h) - localRadius;
        let d = select(segmentDistance, roundDabDistance, isRoundDab);
        // fwidth(d) is how much d changes per screen pixel at this
        // fragment — using it as the step width keeps the edge a crisp,
        // constant ~1px transition at any zoom or supersample factor,
        // instead of a fixed constant that can end up too wide.
        let aa = max(fwidth(d), 1e-4);
        let cov = clamp(0.5 - d / aa, 0.0, 1.0);
        // An SDF step overestimates circles smaller than one backing pixel:
        // a near-zero circle centered on a fragment can still return ~50%
        // coverage. Scale tiny dabs by their physical circle area.
        let circleArea = min(1.0, 3.14159265 * localRadius * localRadius);
        let strokeWidth = min(1.0, 2.0 * localRadius);
        let subpixelArea = select(strokeWidth, circleArea, isRoundDab);
        // The stroke target is a one-channel coverage mask. Maximum blending
        // keeps the strongest coverage and rejects redundant interior overdraw.
        return vec4f(in.alpha * cov * subpixelArea, 0.0, 0.0, 1.0);
      }
    `});

    const strokePipeline = device.createRenderPipeline({
      layout: 'auto',
      vertex: {

          attributes: [

            { format: 'float32x2', offset: 8,  shaderLocation: 1 }, // p0
            { format: 'float32x2', offset: 16, shaderLocation: 2 }, // p1
            { format: 'float32',   offset: 24, shaderLocation: 3 }, // r0
            { format: 'float32',   offset: 28, shaderLocation: 4 }, // r1
            { format: 'float32',   offset: 32, shaderLocation: 5 }, // alpha

      struct VSOut { @builtin(position) pos: vec4f, @location(0) uv: vec2f };
      @vertex
      fn vs(@builtin(vertex_index) i: u32) -> VSOut {
        var p = array<vec2f,3>(vec2f(-1.0,-1.0), vec2f(3.0,-1.0), vec2f(-1.0,3.0));

        // average coverage of all 16 supersampled paint pixels.
        let sourceOrigin = vec2i(in.pos.xy) * 4;

      }
    `});

    `});

    // Margin, in backing-store pixels, added around each shape's true
    // geometry when building its bounding quad. This just needs to be big
    // enough that the ~1px antialiased edge (computed for real in the
    // fragment shader, see strokeShader above) is fully inside the quad —
    // it does NOT control how soft the edge looks anymore.
    const AA_MARGIN = 2.0;

    // Build a bounding quad (2 triangles) for a round dab. The true circle
    // is defined by (cx,cy,r), carried per-vertex and resolved exactly in
    // the fragment shader — a dab is just a zero-length capsule (p0 == p1).
    function circleVerts(cx, cy, r, alpha){
      const half = r + AA_MARGIN;
      const x0 = cx-half, y0 = cy-half, x1 = cx+half, y1 = cy+half;
      const v = (x,y) => [x,y, cx,cy, cx,cy, r, r, alpha];
      const out = [];
      out.push.apply(out, v(x0,y0)); out.push.apply(out, v(x1,y0)); out.push.apply(out, v(x1,y1));
      out.push.apply(out, v(x0,y0)); out.push.apply(out, v(x1,y1)); out.push.apply(out, v(x0,y1));
      return out;
    }
    // Build a bounding quad (2 triangles) for the capsule between two
    // points — the straight part of a line segment. The true capsule is
    // carries independent start/end radii, producing a continuous tapered
    // SDF without rounded radius-step joints.
    function segmentVerts(x0,y0,x1,y1,r0,r1,alpha){
      const dx = x1-x0, dy = y1-y0;
      const len = Math.hypot(dx,dy) || 1;
      const ux = dx/len, uy = dy/len;      // along-segment unit vector
      const nx = -uy, ny = ux;             // normal unit vector
      const maxR = Math.max(r0, r1);
      const ext = maxR + AA_MARGIN;        // extend past each endpoint to cover both rounded caps
      const hw = maxR + AA_MARGIN;         // cover the widest end of the tapered segment
      const p0 = { x: x0-ux*ext, y: y0-uy*ext };
      const p1 = { x: x1+ux*ext, y: y1+uy*ext };
      const c0 = { x: p0.x+nx*hw, y: p0.y+ny*hw };
      const c1 = { x: p1.x+nx*hw, y: p1.y+ny*hw };
      const c2 = { x: p1.x-nx*hw, y: p1.y-ny*hw };
      const c3 = { x: p0.x-nx*hw, y: p0.y-ny*hw };
      const v = (p) => [p.x,p.y, x0,y0, x1,y1, r0,r1,alpha];
      const out = [];
      out.push.apply(out, v(c0)); out.push.apply(out, v(c1)); out.push.apply(out, v(c2));
      out.push.apply(out, v(c0)); out.push.apply(out, v(c2)); out.push.apply(out, v(c3));
      return out;
    }

    let prevRaw = null;   // point two samples ago (curve control point source)

    let lastMid = null;   // last curve endpoint actually drawn (midpoint-based)

    let lastAlpha = null; // alpha actually used for the most recent segment (tracks pressure taper)

                              // so endStroke shouldn't stamp a second one on top of it
    // Clean-room approximation of the documented Moving Average mode.
    // The UI value is the literal window Amount: 0 disables smoothing;
    // 7, 30, and 69 match the investigated preset values. All retained
    // positions have equal weight. This is an original hypothesis, not a
    // claim about Lazy Nezumi's proprietary implementation.
    let smoothBuf = [];
    let lastMoveEventTime = 0;
    let pressureBuf = [];
    let delayedPressure = 0;
    // Mirrors pushSmoothBuf's windowing exactly, but for pressure instead of
    // position. This gives a "stabilized pressure" value that corresponds to
    // the delayed brush position (lastRaw), not the live/raw pen reading —
    // the same way lastRaw lags behind the real pointer position.
    function pushPressureBuf(p){
      const maxLen = movingAverageAmount();
      pressureBuf.push(p);
      while(pressureBuf.length > maxLen) pressureBuf.shift();
      if(maxLen === 1) return p;
      let s = 0;
      for(const v of pressureBuf) s += v;
      return s / pressureBuf.length;
    }
    // Dequeue-only companion to pushPressureBuf: used ONLY during Moving
    // Release finish (see endStroke). It never injects a new sample — it
    // just lets the buffer's already-recorded real pressure history keep
    // aging out one sample per tick, the same way pushSmoothBuf's position
    // window keeps shifting. This is what reproduces a genuine pointed
    // ending on a fast flick: whatever real decaying pressure was already
    // captured during drawing continues to play out, never a frozen
    // scalar and never a fabricated envelope.
    function drainPressureBuf(){
      if(pressureBuf.length > 1) pressureBuf.shift();
      let s = 0;
      for(const v of pressureBuf) s += v;
      return s / pressureBuf.length;
    }
    let lastInputRaw = null;
    let lastInputPressure = 0;

    // --- Contact pressure tracking --------------------------------------
    // Some pen/tablet stacks emit one or more pointermove samples with
    // pressure decaying toward 0 as the tip leaves the surface, *before*
    // pointerup actually fires — these are release artifacts, not real
    // drawing input. lastContactPressure only updates from samples above a
    // small contact floor, so it always reflects genuine "pen down and
    // pressing" pressure, never a trailing near-zero sample and never the
    // pointerup event's own (usually 0) pressure value.
    const CONTACT_PRESSURE_FLOOR = 0.02;
    let lastContactPressure = 0;

    // Debug/validation hook (goal: "test with position-only debug mode").
    // window.setStabDebugConstantPressure(true) forces every sample's
    // pressure — including finish — to a fixed 0.7, isolating position
    // catch-up from pressure handling for testing.
    let DEBUG_CONSTANT_PRESSURE = false;
    window.setStabDebugConstantPressure = (on) => {
      DEBUG_CONSTANT_PRESSURE = !!on;
      console.log('[stabilizer debug] constant pressure override:', DEBUG_CONSTANT_PRESSURE);
    };
    function effectivePressure(p){
      return DEBUG_CONSTANT_PRESSURE ? 0.7 : p;
    }

    // NOTE: there is intentionally no release-envelope / taper-curve
    // function here. The finish stage (see endStroke below) replays the
    // frozen pointer-up target through the same pushSmoothBuf/
    // pushPressureBuf moving-average buffers used during normal drawing,
    // symmetrically, tick for tick. Whatever pressure shape appears at
    // the very end of a stroke is just that real buffer converging —
    // never a synthesized fade.

    // --- Finish-stage velocity tracking --------------------------------
    // Tracks the actual (unstabilized) pointer speed during normal drawing
    // so the accelerated finish (below) can tell a fast flick from a slow,
    // deliberate stroke. This only records metadata alongside existing
    // sample processing — it doesn't change curve reconstruction, spacing,
    // or what gets drawn during a normal stroke.
    let recentSpeedPxMs = 0;
    let lastVelRaw = null;
    let lastVelTime = 0;
    function trackVelocity(raw, timeStamp){
      const now = Number.isFinite(timeStamp) ? timeStamp : performance.now();
      if(lastVelRaw && lastVelTime){
        const dt = now - lastVelTime;
        if(dt > 0){
          const dist = Math.hypot(raw.x - lastVelRaw.x, raw.y - lastVelRaw.y);
          const inst = dist / dt;
          // Light exponential smoothing so one noisy sample can't spike it.
          recentSpeedPxMs = recentSpeedPxMs * 0.7 + inst * 0.3;
        }
      }
      lastVelRaw = raw;
      lastVelTime = now;
    }

    // --- Release-velocity rolling window (Moving/Flick Release taper) ---
    // Separate from recentSpeedPxMs above: that one is an exponential
    // smoothing of instantaneous inter-sample speed, useful for the
    // stationary-vs-moving classification. The flick taper instead wants
    // a genuine windowed distance/time measurement over the last ~40-80ms
    // of *raw* pen movement, so a single noisy final sample can't dominate
    // it and a brief stutter mid-flick can't be mistaken for a slow lift.
    const RELEASE_VEL_WINDOW_MS = 80;   // keep raw samples covering up to this much history
    let releaseVelWindow = []; // {x,y,t} raw (unstabilized) canvas-space samples
    let smoothedReleaseVelocityCanvasPxMs = 0; // lightly smoothed, canvas backing-pixel units
    function pushReleaseVelWindow(raw, timeStamp){
      const t = Number.isFinite(timeStamp) ? timeStamp : performance.now();
      releaseVelWindow.push({ x: raw.x, y: raw.y, t });
      while(releaseVelWindow.length > 1 && t - releaseVelWindow[0].t > RELEASE_VEL_WINDOW_MS){
        releaseVelWindow.shift();
      }
      if(releaseVelWindow.length >= 2){
        const newest = releaseVelWindow[releaseVelWindow.length - 1];
        const oldest = releaseVelWindow[0];
        const dt = Math.max(1, newest.t - oldest.t);
        const dist = Math.hypot(newest.x - oldest.x, newest.y - oldest.y);
        const instWindowed = dist / dt;
        // Light exponential smoothing so one noisy event can't create a
        // large taper on its own.
        smoothedReleaseVelocityCanvasPxMs = smoothedReleaseVelocityCanvasPxMs * 0.7 + instWindowed * 0.3;
      }
    }
    // Converts a canvas backing-pixel value into true screen-space pixels,
    // so flick-strength thresholds and taper lengths behave consistently
    // regardless of zoom level (screenToCanvas divides by zoom and
    // multiplies by SS to get canvas/backing space; this undoes that).
    function canvasPxToScreenPx(canvasPx){
      return canvasPx * zoom / SS;
    }
    function screenPxToCanvasPx(screenPx){
      return screenPx * SS / zoom;
    }

    function movingAverageAmount(){
      if(stabilization <= 0) return 1;
      return Math.max(2, Math.round(stabilization * 100));
    }
    function pushSmoothBuf(p){
      const maxLen = movingAverageAmount();
      smoothBuf.push({ x: p.x, y: p.y });
      while(smoothBuf.length > maxLen) smoothBuf.shift();
      if(maxLen === 1) return { x: p.x, y: p.y };
      let sx = 0, sy = 0;
      for(const pt of smoothBuf){ sx += pt.x; sy += pt.y; }
      return { x: sx / smoothBuf.length, y: sy / smoothBuf.length };
    }
    // ---- Stabilizer leash visualization -------------------------------
    // Draws the lazybrush.dulnan.net-style indicator: a dashed line from
    // the raw pointer position (the "anchor" dot) to the delayed/smoothed
    // brush tip currently being painted, so the stabilization lag is
    // visible in real time instead of being an invisible internal buffer.
    function clearStabViz(){
      stabVizCtx.clearRect(0, 0, stabViz.width, stabViz.height);
    }
    function drawStabilizerLeash(){
      clearStabViz();
      if(!leashEnabled || !drawing || stabilization <= 0) return;
      if(!lastRaw || !lastInputRaw) return;
      const tip = canvasToWrap(lastRaw.x, lastRaw.y);       // stabilized brush position
      const anchor = canvasToWrap(lastInputRaw.x, lastInputRaw.y); // raw pointer position
      const dist = Math.hypot(anchor.x - tip.x, anchor.y - tip.y);
      if(dist < 1) return; // pen sitting exactly on the brush tip — nothing to show

      const ctx = stabVizCtx;
      ctx.save();
      ctx.lineCap = 'round';

      // Dashed leash line connecting anchor -> tip.
      ctx.beginPath();
      ctx.moveTo(anchor.x, anchor.y);
      ctx.lineTo(tip.x, tip.y);
      ctx.setLineDash([5, 5]);
      ctx.lineDashOffset = 0;
      ctx.lineWidth = 1.25;
      ctx.strokeStyle = 'rgba(20,20,20,0.55)';
      ctx.stroke();
      ctx.setLineDash([]);

      // Anchor dot — the true, unstabilized pointer position.
      ctx.beginPath();
      ctx.arc(anchor.x, anchor.y, 4, 0, Math.PI * 2);
      ctx.fillStyle = '#141414';
      ctx.fill();
      ctx.lineWidth = 1.5;
      ctx.strokeStyle = '#ffffff';
      ctx.stroke();

      // Brush-tip dot — where the stabilized stroke is actually landing.
      ctx.beginPath();
      ctx.arc(tip.x, tip.y, 3, 0, Math.PI * 2);
      ctx.fillStyle = '#5aa9ff';
      ctx.fill();
      ctx.lineWidth = 1.25;
      ctx.strokeStyle = '#ffffff';
      ctx.stroke();

      ctx.restore();
    }

    beginStroke = function(e){
      if(chaseRAF){ cancelAnimationFrame(chaseRAF); chaseRAF = null; }
      const p = screenToCanvas(e.clientX, e.clientY);
      pushUndo();
      beginStrokeCoverage();
      drawing = true;
      prevRaw = p;
      lastRaw = p;
      lastMid = p;
      strokeMoved = false;
      __feedPointCallIndex = 0; // reset stroke-start diagnostic counter (see feedPoint)
      const amount = movingAverageAmount();
      // Prefill both buffers to the full moving-average window length at
      // pointer-down, instead of starting from a single sample. Without
      // this, the first N samples of a stroke get averaged over a buffer
      // that is still filling up (1 sample, then 2, then 3, ... up to
      // `amount`), so stabilization strength ramps from "off" to "full"
      // over the first N points instead of being constant from the very
      // first move. Each slot is created via its own `() => ({...})` call
      // (not a shared object literal), so later in-place mutation of one
      // buffer entry can never alias another.
      smoothBuf = Array.from({ length: amount }, () => ({ x: p.x, y: p.y }));
      lastInputRaw = p;
      const startPressure = effectivePressure(getPressure(e));
      pressureBuf = Array.from({ length: amount }, () => startPressure);
      delayedPressure = startPressure;
      lastInputPressure = startPressure;
      lastContactPressure = startPressure;
      recentSpeedPxMs = 0;
      lastVelRaw = p;
      lastVelTime = performance.now();
      lastMoveEventTime = performance.now();
      releaseVelWindow = [{ x: p.x, y: p.y, t: performance.now() }];
      smoothedReleaseVelocityCanvasPxMs = 0;
      const r = pressureRadius(getBrushSize(), startPressure);
      lastR = r;
      const a = pressureAlpha(startPressure);
      lastAlpha = a;
      const verts = circleVerts(p.x, p.y, r, a);
      drawVertsIntoAcc(new Float32Array(verts), verts.length / 9, colorRgb, tool === 'eraser');
      if(strokeHoldRAF === null){
        strokeHoldLastT = 0;
        strokeHoldRAF = requestAnimationFrame(strokeHoldTick);
      }
    };

    function drawQuadCurve(all, p0, control, p1, r0, r1, alpha, steps){
      let prevPt = p0;
      let prevR = r0;
      for(let i=1;i<=steps;i++){
        const t = i/steps;
        const mt = 1-t;
        const x = mt*mt*p0.x + 2*mt*t*control.x + t*t*p1.x;
        const y = mt*mt*p0.y + 2*mt*t*control.y + t*t*p1.y;
        const r = r0 + (r1-r0)*t;
        all.push.apply(all, segmentVerts(prevPt.x, prevPt.y, x, y, prevR, r, alpha));
        prevPt = {x,y};
        prevR = r;
      }
      return prevPt;
    }

    // Feeds one point through the midpoint-quadratic curve fitter and
    // advances lastMid/prevRaw/lastRaw. Factored out so moveStroke can call
    // it multiple times per real sample (see subdivision below).
    // TEMPORARY DIAGNOSTIC (pairing verification only — remove after confirming):
    // Records every feedPoint call so the final one (immediately before
    // pointerup) can be inspected to confirm raw/pressure/radius all came
    // from the same tick, and that delayedPressure at call time matches
    // the pressure argument actually passed in.
    // TEMPORARY DIAGNOSTIC (stroke-start stabilization investigation only —
    // remove once the startup-kink question is settled). Logs p0/control/p1,
    // buffer length, the stabilized output point, and the tangent direction
    // for the first three feedPoint() calls of each stroke, so the first
    // (potentially degenerate) curve segment can be inspected directly.
    let __feedPointCallIndex = 0;
    function feedPoint(all, raw, pressure){
      strokeMoved = true;
      const r = pressureRadius(getBrushSize(), pressure);
      const a = pressureAlpha(pressure);
      const midStart = lastMid;
      const midEnd = { x: (lastRaw.x + raw.x) / 2, y: (lastRaw.y + raw.y) / 2 };
      const startR = lastR == null ? r : lastR;
      if(__feedPointCallIndex < 3){
        const tangent = { x: midEnd.x - midStart.x, y: midEnd.y - midStart.y };
        const tangentLen = Math.hypot(tangent.x, tangent.y);
        console.log('[feedPoint diag]', {
          call: __feedPointCallIndex,
          p0: { x: midStart.x, y: midStart.y },
          control: { x: lastRaw.x, y: lastRaw.y },
          p1: midEnd,
          smoothBufLength: smoothBuf.length,
          stabilizedOutput: raw,
          tangentDirection: tangentLen > 1e-6
            ? { x: tangent.x / tangentLen, y: tangent.y / tangentLen }
            : 'degenerate (p0 == p1)'
        });
      }
      __feedPointCallIndex++;
      lastMid = drawQuadCurve(all, midStart, lastRaw, midEnd, startR, r, a, 10);
      lastR = r;
      lastAlpha = a;
      prevRaw = lastRaw;
      lastRaw = raw;
    }

    function processStrokeBatch(e){
      if(!drawing) return;
      const t0 = performance.now();
      const events = (typeof e.getCoalescedEvents === 'function' && e.getCoalescedEvents().length)
        ? e.getCoalescedEvents() : [e];

      // Long jumps are subdivided only for curve tessellation; this does not change the Moving Average.
      const maxStep = Math.max(2, scaledBrushSize() * 0.6);

      let all = [];
      for(const ev of events){
        const inputRaw = screenToCanvas(ev.clientX, ev.clientY);
        const rawPressure = getPressure(ev);
        const pressure = effectivePressure(rawPressure);
        trackVelocity(inputRaw, ev.timeStamp);
        pushReleaseVelWindow(inputRaw, ev.timeStamp);
        // Hardware (mouse and especially pen/tablet) commonly reports a
        // stream of pointermove events even while the hand is essentially
        // motionless — sub-pixel sensor/OS jitter. Only count this sample as
        // real movement (for held-before-lift detection) if it's larger
        // than that noise floor; otherwise a "held still" pause never
        // actually reaches HOLD_BEFORE_LIFT_MS because jitter keeps
        // resetting the timer every few milliseconds.
        const JITTER_FLOOR_PX = 1.5 * zoomSmoothingFactor();
        if(!lastInputRaw || Math.hypot(inputRaw.x - lastInputRaw.x, inputRaw.y - lastInputRaw.y) > JITTER_FLOOR_PX){
          lastMoveEventTime = performance.now();
        }
        lastInputRaw = inputRaw;
        lastInputPressure = pressure;
        // Only real contact pressure above the floor updates the value used
        // to freeze the finish — a trailing near-zero sample right before
        // release (tip lifting off) must not overwrite it.
        if(pressure > CONTACT_PRESSURE_FLOOR) lastContactPressure = pressure;
        delayedPressure = pushPressureBuf(pressure);
        const raw = pushSmoothBuf(inputRaw);
        const jumpDx = raw.x - lastRaw.x, jumpDy = raw.y - lastRaw.y;
        const jumpDist = Math.hypot(jumpDx, jumpDy);
        if(jumpDist > maxStep){
          const origin = lastRaw;
          const steps = Math.ceil(jumpDist / maxStep);
          for(let i = 1; i <= steps; i++){
            const t = i / steps;
            feedPoint(all, {
              x: origin.x + jumpDx * t,
              y: origin.y + jumpDy * t
            }, delayedPressure);
          }
        } else {
          feedPoint(all, raw, delayedPressure);
        }
      }
      if(all.length){
        const arr = new Float32Array(all);
        drawVertsIntoAcc(arr, arr.length / 9, colorRgb, tool === 'eraser');
      }
      drawStabilizerLeash();
      reportPerf(performance.now() - t0);
    }

    // Pointer events only collect samples. Rendering happens at most once per
    // display frame, so coalesced mouse input is presented as one continuous
    // curve update instead of a burst of event-timed canvas updates.
    let pendingStrokeSamples = [];
    let strokeFrameRAF = null;
    function flushPendingStrokeSamples(){
      if(strokeFrameRAF !== null){
        cancelAnimationFrame(strokeFrameRAF);
        strokeFrameRAF = null;
      }
      if(!drawing || pendingStrokeSamples.length === 0) return;
      const samples = pendingStrokeSamples;
      pendingStrokeSamples = [];
      processStrokeBatch({ getCoalescedEvents(){ return samples; } });
    }
    moveStroke = function(e){
      const source = (typeof e.getCoalescedEvents === 'function' && e.getCoalescedEvents().length)
        ? e.getCoalescedEvents() : [e];
      for(const ev of source){
        pendingStrokeSamples.push({
          clientX: ev.clientX,
          clientY: ev.clientY,
          pointerType: ev.pointerType || e.pointerType,
          pressure: typeof ev.pressure === 'number' ? ev.pressure : e.pressure
        });
      }
      // The persistent catch-up loop (started in beginStroke) already drives
      // per-frame processing, so no per-event RAF scheduling is needed here.
    };

    // Persistent per-frame loop that runs for the whole duration the pointer
    // is down. When real samples arrived this frame, they're processed as
    // before. When the pen is simply being held still, there are no new
    // pointer events to drain the Moving Average buffer, so we keep feeding
    // the last known position back into it every frame — this is what lets
    // the stabilized stroke visibly "catch up" to a stationary held pen
    // instead of only catching up once the pen lifts.
    let strokeHoldRAF = null;
    let strokeHoldLastT = 0;
    function strokeHoldTick(t){
      strokeHoldRAF = null;
      if(!drawing) return;
      if(pendingStrokeSamples.length){
        flushPendingStrokeSamples();
        strokeHoldLastT = t;
      } else if(lastInputRaw && lastRaw){
        // Drive the catch-up by elapsed real time rather than an assumed
        // frame count. A fixed "N slots per frame" approach looks choppy
        // whenever the actual frame interval drifts from 60fps (background
        // tab throttling, dropped frames, high/low refresh displays, etc.),
        // because the on-screen speed of the catch-up changes frame to
        // frame. Computing ticks from dt keeps the drain rate constant in
        // real time, so the motion stays smooth regardless of frame pacing.
        const now = (typeof t === 'number') ? t : performance.now();
        const dt = strokeHoldLastT ? Math.min(now - strokeHoldLastT, 100) : 16.7; // clamp big gaps (tab switches)
        strokeHoldLastT = now;
        // No real pointer samples arrived this frame — the hand is
        // genuinely idle, not just between events, so ease the tracked
        // velocity toward zero. Without this, a pointer that stops dead
        // (no jitter at all) leaves recentSpeedPxMs frozen at its last
        // in-motion value, which would wrongly still read as "moving" if
        // the pen lifts right after this pause.
        recentSpeedPxMs *= Math.max(0, 1 - dt / 120);
        const catchUpDurationMs = 350; // time to fully converge at the current amount
        const ticksPerMs = movingAverageAmount() / catchUpDurationMs;
        const ticks = Math.max(1, Math.round(dt * ticksPerMs));
        const all = [];
        let raw = null;
        for(let i = 0; i < ticks; i++){
          raw = pushSmoothBuf(lastInputRaw);
          const dx = raw.x - lastRaw.x, dy = raw.y - lastRaw.y;
          if(Math.hypot(dx, dy) <= 1e-4) break;
          feedPoint(all, raw, delayedPressure);
        }
        if(all.length){
          const arr = new Float32Array(all);
          drawVertsIntoAcc(arr, arr.length / 9, colorRgb, tool === 'eraser');
        }
        drawStabilizerLeash();
      } else {
        strokeHoldLastT = t;
      }
      strokeHoldRAF = requestAnimationFrame(strokeHoldTick);
    }

    function resetStrokeState(){
      prevRaw = null;
      lastRaw = null;
      lastMid = null;
      lastR = null;
      lastAlpha = null;
      smoothBuf = [];
      pressureBuf = [];
      lastInputRaw = null;
      lastInputPressure = 0;
      lastContactPressure = 0;
      lastVelRaw = null;
      lastVelTime = 0;
      recentSpeedPxMs = 0;
    }

    function logFinishDiagnostics(diag){
      console.log('[stabilizer finish]', {
        mode: diag.wasStoppedBeforeLift ? 'STATIONARY_HOLD' : 'MOVING_RELEASE',
        stationaryDurationMs: Math.round(diag.stationaryDurationMs),
        recentSpeedPxMsAtLift: +diag.recentSpeedPxMsAtLift.toFixed(4),
        finishDurationMs: Math.round(diag.finishDurationMs),
        maxCatchUpSpeedPxPerMs: +diag.maxCatchUpSpeedPxMs.toFixed(3),
        startGapPx: +diag.startDist.toFixed(2),
        endpointErrorPx: +diag.endpointError.toFixed(4),
        lastContactPressure: +diag.lastContactPressure.toFixed(3),
        pointerupRawPressure: diag.pointerupRawPressure,
        finishPressure: +diag.finishPressure.toFixed(3),
        recentVelocityPxMs: diag.recentVelocityPxMs != null ? +diag.recentVelocityPxMs.toFixed(4) : null,
        flickStrength: diag.flickStrength != null ? +diag.flickStrength.toFixed(3) : null,
        taperLengthPx: diag.taperLengthPx != null ? +diag.taperLengthPx.toFixed(2) : null,
        finalEmittedPressure: +diag.finalEmittedPressure.toFixed(3),
        finalDabRadius: +diag.finalDabRadius.toFixed(2),
        finalDabAlpha: +diag.finalDabAlpha.toFixed(3),
        ticksEmitted: diag.ticksEmitted
      });
    }

    let finishRAF = null;

    endStroke = function(e){
      if(!drawing) return;
      if(strokeHoldRAF !== null){
        cancelAnimationFrame(strokeHoldRAF);
        strokeHoldRAF = null;
      }
      if(finishRAF !== null){
        cancelAnimationFrame(finishRAF);
        finishRAF = null;
      }

      // ---- Capture finish pressure from the brush tip, not the anchor ----
      // Must happen AFTER flushPendingStrokeSamples(): on a fast stroke,
      // several real pointermove samples are often still queued at the
      // moment pointerup fires (coalesced/batched, rendered once per
      // frame). Those are genuine drawing data — flushing them lets any
      // real trailing pressure decrease (fast lift-off) show up honestly.
      // This is still safe from anchor/pointer-up contamination: pointerup
      // never calls moveStroke, so its own (often near-zero) pressure
      // value never enters processStrokeBatch/pressureBuf either way —
      // capturing after flush only picks up real queued samples, never
      // the pointerup event itself.
      flushPendingStrokeSamples();
      const finishPressure = delayedPressure;
      drawing = false;

      if(!(strokeMoved && lastMid && lastRaw)){
        clearStabViz();
        resetStrokeState();
        return;
      }

      const endpoint = e && Number.isFinite(e.clientX) && Number.isFinite(e.clientY)
        ? screenToCanvas(e.clientX, e.clientY) : lastInputRaw;

      // ---- Two finish modes -------------------------------------------
      // Mode A (Stationary Hold): the raw pen genuinely stopped before
      // pointer-up while the stabilized tip was still catching up.
      // Pressure freezes at finishPressure (captured above) — constant,
      // rounded finish, no taper.
      // Mode B (Moving Release): the pen was still moving at pointer-up.
      // Pressure must NOT freeze — it continues draining from its own
      // already-recorded real history (drainPressureBuf), reproducing the
      // natural pointed ending a fast flick had before any of this finish
      // logic ever ran.
      //
      // Classification uses rolling/smoothed signals, not the final
      // sample alone, so a brief slowdown mid-flick can't be mistaken for
      // a genuine stop:
      //  - recentSpeedPxMs is already exponentially smoothed (see
      //    trackVelocity) and only decays toward zero over real elapsed
      //    idle time (see strokeHoldTick) — one slow sample can't collapse it.
      //  - stationaryDurationMs comes from lastMoveEventTime, which only
      //    advances when a real sample exceeds the jitter floor — so it
      //    reflects genuine dwell, not per-event noise.
      const pointerupRawPressure = e && typeof e.pressure === 'number' ? e.pressure : null;

      const startDist = endpoint ? Math.hypot(endpoint.x - lastRaw.x, endpoint.y - lastRaw.y) : 0;

      const HOLD_SPEED_THRESHOLD = 0.03; // px/ms (~30 px/s) — effectively stopped
      const HOLD_DWELL_MS = 150;         // must have been still for this long
      const MIN_CATCHUP_GAP = 1.5;       // px — brush tip must still be meaningfully behind
      const stationaryDurationMs = performance.now() - lastMoveEventTime;
      const brushToAnchorGapPx = startDist;
      const wasStoppedBeforeLift =
        stationaryDurationMs >= HOLD_DWELL_MS &&
        recentSpeedPxMs <= HOLD_SPEED_THRESHOLD &&
        brushToAnchorGapPx >= MIN_CATCHUP_GAP;

      // ---- Moving/Flick Release: velocity-aware taper --------------------
      // Only meaningful for Mode B (Moving Release) — Stationary Hold keeps
      // its frozen constant-width finish untouched below. Everything here
      // is geometric (driven by travelled finish-path distance), not a
      // time-based fade, per the spec: a faster flick must produce a
      // longer, more visible pressure falloff, not just a quicker one.
      function smoothstep01(x){
        const t = Math.max(0, Math.min(1, x));
        return t * t * (3 - 2 * t);
      }
      let taperLengthCanvasPx = 0;
      let flickStrength = 0;
      let recentVelocityPxMs = 0;
      if(!wasStoppedBeforeLift){
        // Recent raw pen speed (screen-space, zoom-independent), from the
        // rolling window fed during normal drawing — not the pointerup
        // event alone, which may be noisy or already slowing down.
        recentVelocityPxMs = canvasPxToScreenPx(smoothedReleaseVelocityCanvasPxMs);

        const minFlickSpeed = 0.15; // px/ms (~150 px/s) — barely a flick
        const maxFlickSpeed = 1.6;  // px/ms (~1600 px/s) — a hard flick
        flickStrength = smoothstep01(
          (recentVelocityPxMs - minFlickSpeed) / (maxFlickSpeed - minFlickSpeed)
        );

        const slowTaperLengthPx = 6;   // screen px — short, subtle taper
        const fastTaperLengthPx = 70;  // screen px — long, clearly visible taper
        let taperLengthPx = slowTaperLengthPx + (fastTaperLengthPx - slowTaperLengthPx) * flickStrength;

        // Taper length may also scale modestly with brush diameter, but
        // never dominate — velocity remains the primary driver.
        const sizeFactor = 0.12;
        taperLengthPx += getBrushSize() * sizeFactor;

        const MIN_TAPER_PX = 4, MAX_TAPER_PX = 140;
        taperLengthPx = Math.max(MIN_TAPER_PX, Math.min(MAX_TAPER_PX, taperLengthPx));

        taperLengthCanvasPx = screenPxToCanvasPx(taperLengthPx);
      }
      // Monotonic, smooth release envelope: 1 at the start of the taper,
      // 0 exactly at the true endpoint. progress is travelled finish-path
      // distance / taperLengthCanvasPx, clamped to [0,1] — geometric, not
      // time-based, so a fast flick's longer taperLengthCanvasPx produces a
      // visibly longer narrowing rather than just a faster fade.
      function releaseEnvelope(progress){
        return 1 - smoothstep01(progress);
      }

      if(!endpoint || startDist < 0.15){
        // Nothing meaningful to catch up on (stabilization off, or the pen
        // lifted essentially where it was already drawn to).
        const all = [];
        if(endpoint) feedPoint(all, endpoint, finishPressure);
        if(all.length){
          const arr = new Float32Array(all);
          drawVertsIntoAcc(arr, arr.length / 9, colorRgb, tool === 'eraser');
        }
        clearStabViz();
        resetStrokeState();
        return;
      }

      const diag = {
        startTime: performance.now(),
        startDist,
        finishPressure,
        pointerupRawPressure,
        lastContactPressure,
        wasStoppedBeforeLift,
        stationaryDurationMs,
        recentSpeedPxMsAtLift: recentSpeedPxMs,
        recentVelocityPxMs,
        flickStrength,
        taperLengthPx: taperLengthCanvasPx ? canvasPxToScreenPx(taperLengthCanvasPx) : 0,
        maxCatchUpSpeedPxMs: 0,
        finishDurationMs: 0,
        endpointError: 0,
        finalDabRadius: 0,
        finalDabAlpha: 0,
        ticksEmitted: 0,
        frames: [] // per-frame trace: position buffer drain, for validation
      };
      let lastTickTime = diag.startTime;

      // ---- Finish playback pacing (position/timing only) -----------------
      // This controls how FAST the already-existing Moving Average buffers
      // drain after release — not what they drain TO. Small remaining gap
      // -> short, gentle finish. Large flick -> faster finish. This is
      // unchanged in spirit from before; only the pressure side beneath it
      // has changed.
      function smoothstep01(x){
        const t = Math.max(0, Math.min(1, x));
        return t * t * (3 - 2 * t);
      }
      function mix(a, b, t){ return a + (b - a) * t; }
      const MIN_FINISH_MS = 80;
      const MAX_FINISH_MS = 260;
      const gapNorm = Math.max(0, Math.min(1, startDist / 400)); // 400px ~= "large" gap
      const speedNorm = Math.max(0.5, Math.min(3, 0.6 + recentSpeedPxMs * 1.2));
      let targetDurationMs = mix(MIN_FINISH_MS, MAX_FINISH_MS, gapNorm) / speedNorm;
      targetDurationMs = Math.max(MIN_FINISH_MS, Math.min(MAX_FINISH_MS, targetDurationMs));
      diag.targetDurationMs = targetDurationMs;

      const totalTicksNeeded = movingAverageAmount();
      const avgTicksPerMs = totalTicksNeeded / targetDurationMs;
      const startRatePerMs = avgTicksPerMs * 0.6;
      const endRatePerMs = avgTicksPerMs * 1.6;

      // Accelerated Finish Line. Every tick still advances the exact same
      // position moving-average buffer used during normal drawing
      // (pushSmoothBuf) — this is what keeps the finish following the real
      // stabilized trajectory instead of a shortcut straight line.
      // Pressure source depends on wasStoppedBeforeLift (computed above):
      //  - Stationary Hold: finishPressure, frozen, same value every tick.
      //  - Moving Release: drainPressureBuf(), which lets the already-
      //    recorded real pressure history keep aging out one sample per
      //    tick — no injection, no freeze, reproduces the natural pointed
      //    ending a fast flick had before finish-phase logic existed.
      // What changes with playback speed is only the position drain RATE
      // (ticksPerMs); the pressure SOURCE is fixed per-stroke by the mode.
      let tailAlreadyDrawn = false; // guards against a second, disconnected-looking emit
                                     // once the exact endpoint sample has been drawn

      // ---- Moving Release taper placement (velocity controls WHERE the ----
      // taper sits on the remaining path, not WHETHER the path finishes).
      // The brush tip must always reach the anchor, TVPaint-style — so the
      // envelope is compressed into the last taperLengthCanvasPx of the
      // catch-up path: full finishPressure until travelled reaches
      // taperStartDist, then a smooth fall to exactly 0 at travelled ===
      // startDist (the anchor).
      // taperStartDist = max(0, startDist - taperLengthCanvasPx):
      //  - fast flick -> longer taperLengthCanvasPx -> smaller
      //    taperStartDist -> taper begins earlier and covers more of the
      //    path -> longer, more visible narrowing.
      //  - slow release -> short taperLengthCanvasPx -> taperStartDist
      //    close to startDist -> taper only occupies the final sliver of
      //    the path, starting late and finishing almost immediately.
      // Either way progress reaches exactly 1 (envelope 0) only when
      // travelled reaches startDist — i.e. exactly when the brush tip
      // physically arrives at the anchor.
      const taperStartDistCanvasPx = wasStoppedBeforeLift
        ? startDist // unused for Stationary Hold, but keeps the var defined
        : Math.max(0, startDist - taperLengthCanvasPx);
      function movingReleasePressureAt(travelled){
        const span = Math.max(1e-4, startDist - taperStartDistCanvasPx);
        const progress = Math.min(1, Math.max(0, (travelled - taperStartDistCanvasPx) / span));
        return finishPressure * releaseEnvelope(progress);
      }

      function finishTick(t){
        finishRAF = null;
        const now = Number.isFinite(t) ? t : performance.now();
        const dt = Math.min(now - lastTickTime, 60);
        lastTickTime = now;

        const remaining = Math.hypot(endpoint.x - lastRaw.x, endpoint.y - lastRaw.y);

        if(remaining <= 0.15 || smoothBuf.length === 0){
          if(!tailAlreadyDrawn){
            // Final sample: position always lands exactly on the true
            // endpoint/anchor, for both modes. Pressure source:
            //  - Stationary Hold: frozen finishPressure.
            //  - Moving Release: the compressed envelope evaluated at
            //    travelled === startDist, which is 0 by construction —
            //    the taper's own natural end, not a forced/artificial zero,
            //    and it lands exactly where the brush tip lands.
            const finalPressure = wasStoppedBeforeLift ? finishPressure : movingReleasePressureAt(startDist);
            const all = [];
            feedPoint(all, endpoint, finalPressure);
            if(all.length){
              const arr = new Float32Array(all);
              drawVertsIntoAcc(arr, arr.length / 9, colorRgb, tool === 'eraser');
            }
            diag.endpointError = Math.hypot(endpoint.x - lastRaw.x, endpoint.y - lastRaw.y);
            diag.finishDurationMs = now - diag.startTime;
            diag.finalDabRadius = pressureRadius(getBrushSize(), finalPressure);
            diag.finalDabAlpha = pressureAlpha(finalPressure);
            diag.finalEmittedPressure = finalPressure;
            logFinishDiagnostics(diag);
          }
          clearStabViz();
          resetStrokeState();
          return;
        }

        const elapsed = now - diag.startTime;
        const timeProgress = Math.max(0, Math.min(1, elapsed / targetDurationMs));
        const ticksPerMs = mix(startRatePerMs, endRatePerMs, smoothstep01(timeProgress));
        const maxTicks = movingAverageAmount();
        const ticks = Math.max(1, Math.min(maxTicks, Math.round(dt * ticksPerMs)));

        const all = [];
        for(let i = 0; i < ticks; i++){
          const caught = pushSmoothBuf(endpoint);
          const stepDist = Math.hypot(caught.x - lastRaw.x, caught.y - lastRaw.y);
          if(stepDist <= 1e-4) break;

          const stepSpeed = stepDist / Math.max(dt / ticks, 0.001);
          if(stepSpeed > diag.maxCatchUpSpeedPxMs) diag.maxCatchUpSpeedPxMs = stepSpeed;
          diag.ticksEmitted++;

          // Pressure source depends on the finish mode determined once at
          // pointer-up (wasStoppedBeforeLift): frozen constant for
          // Stationary Hold. For Moving Release, the envelope is a function
          // of travelled finish-path distance (not elapsed time), always
          // reaching 0 exactly when travelled === startDist — i.e. exactly
          // when the brush tip arrives at the anchor. Nothing ever stops
          // being fed to feedPoint before the anchor is reached.
          let emittedPressure;
          if(wasStoppedBeforeLift){
            emittedPressure = finishPressure;
          } else {
            const remainingNow = Math.hypot(endpoint.x - caught.x, endpoint.y - caught.y);
            const travelled = Math.max(0, startDist - remainingNow);
            emittedPressure = movingReleasePressureAt(travelled);
          }

          if(diag.frames.length < 60){
            diag.frames.push({
              tick: diag.ticksEmitted, timeProgress,
              ticksPerMs: +ticksPerMs.toFixed(4), pressure: +emittedPressure.toFixed(4)
            });
          }

          // Position converges toward the anchor; pressure is the frozen
          // brush-tip value. Same tick, so they can never drift apart.
          feedPoint(all, caught, emittedPressure);
        }
        if(all.length){
          const arr = new Float32Array(all);
          drawVertsIntoAcc(arr, arr.length / 9, colorRgb, tool === 'eraser');
        }
        drawStabilizerLeash();
        finishRAF = requestAnimationFrame(finishTick);
      }

      finishRAF = requestAnimationFrame(finishTick);
    };

  } else {
    // ================= Canvas2D fallback (no WebGPU / no hardware GPU) =================
    const ctx2d = canvas.getContext('2d');
    ctx2d.fillStyle = '#ffffff';
    ctx2d.fillRect(0, 0, BW, BH);

    function snapshot(){ return ctx2d.getImageData(0, 0, BW, BH); }

    pushUndo = function(){
      undoStack.push(snapshot());
      if(undoStack.length > MAX_UNDO) undoStack.shift();
      redoStack.length = 0;
    };
    restore = function(stack, otherStack){
      if(stack.length === 0) return;
      otherStack.push(snapshot());
      ctx2d.putImageData(stack.pop(), 0, 0);
    };
    clearCanvas = function(){
      pushUndo();
      ctx2d.fillStyle = '#ffffff';
      ctx2d.fillRect(0, 0, BW, BH);
    };

    let lastPt = null;

    beginStroke = function(e){
      pushUndo();
      drawing = true;
      const p = screenToCanvas(e.clientX, e.clientY);
      lastPt = p;
      ctx2d.globalAlpha = opacity;
      ctx2d.fillStyle = tool === 'eraser' ? '#ffffff' : color;
      ctx2d.beginPath();
      ctx2d.arc(p.x, p.y, scaledBrushSize() / 2, 0, Math.PI * 2);
      ctx2d.fill();
    };

    moveStroke = function(e){
      if(!drawing || !lastPt) return;
      const events = (typeof e.getCoalescedEvents === 'function' && e.getCoalescedEvents().length)
        ? e.getCoalescedEvents() : [e];
      ctx2d.globalAlpha = opacity;
      ctx2d.strokeStyle = tool === 'eraser' ? '#ffffff' : color;
      ctx2d.lineWidth = scaledBrushSize();
      ctx2d.lineCap = 'round';
      ctx2d.lineJoin = 'round';
      for(const ev of events){
        const p = screenToCanvas(ev.clientX, ev.clientY);
        ctx2d.beginPath();
        ctx2d.moveTo(lastPt.x, lastPt.y);
        ctx2d.lineTo(p.x, p.y);
        ctx2d.stroke();
        lastPt = p;
      }
    };

    endStroke = function(){
      drawing = false;
      lastPt = null;
    };
  }

  // ---------- Pointer events ----------
  stageWrap.addEventListener('pointerdown', (e)=>{
    if(e.button === 1){ startPan(e); return; }
    if(cDown){ startSizeDrag(e); return; }
    if(spaceDown && ctrlDown){ startZoomDrag(e); return; }
    if(spaceDown){ startPan(e); return; }
    if(e.button !== 0 && e.pointerType !== 'pen' && e.pointerType !== 'touch') return;
    // Ignore a second pointer (palm touch, stray click, pen+mouse at once)
    // while a stroke from a different pointer is already in progress — a
    // beginStroke() here would reset the shared lastRaw/lastMid/prevRaw
    // state to this new pointer's position, and the original pointer's
    // next moveStroke would then draw a straight line connecting the old
    // stroke's last point to the new pointer's position: exactly the thin,
    // unwavy "ghost" diagonal line artifact.
    if(drawing && activePointerId !== null && e.pointerId !== activePointerId) return;
    stageWrap.setPointerCapture(e.pointerId);
    activePointerId = e.pointerId;
    previewSuppressedUntilMove = true;
    const wrapRect = stageWrap.getBoundingClientRect();
    hoverX = e.clientX - wrapRect.left;
    hoverY = e.clientY - wrapRect.top;
    hoverPointerType = e.pointerType;
    hoverPressure = getPressure(e);
    hovering = true;
    beginStroke(e);
    updateBrushPreview();
  });

  stageWrap.addEventListener('pointermove', (e)=>{
    const wrapRect = stageWrap.getBoundingClientRect();
    hoverX = e.clientX - wrapRect.left;
    hoverY = e.clientY - wrapRect.top;
    hoverPointerType = e.pointerType;
    hoverPressure = getPressure(e);
    hovering = true;
    if(isSizeDragging){ doSizeDrag(e); return; }
    if(isZoomDragging){ doZoomDrag(e); updateBrushPreview(); return; }
    if(isPanning){ doPan(e); updateBrushPreview(); return; }
    if(drawing && e.pointerId === activePointerId) moveStroke(e);
    updateBrushPreview();
  });

  stageWrap.addEventListener('pointerenter', (e)=>{
    hovering = true;
    previewSuppressedUntilMove = false;
    hoverPointerType = e.pointerType;
    hoverPressure = getPressure(e);
    updateBrushPreview();
  });
  stageWrap.addEventListener('pointerleave', ()=>{
    hovering = false;
    if(!isSizeDragging) hideSizePreview();
  });

  window.addEventListener('pointerup', (e)=>{
    if(isSizeDragging){ stopSizeDrag(); return; }
    if(isZoomDragging){ stopZoomDrag(); return; }
    if(isPanning){ stopPan(); return; }
    if(drawing && e.pointerId !== activePointerId) return;
    endStroke(e);
    activePointerId = null;
    updateBrushPreview();
  });
  window.addEventListener('pointercancel', (e)=>{
    if(drawing && e.pointerId !== activePointerId){ return; }
    endStroke(e);
    activePointerId = null;
    stopPan(); stopZoomDrag(); stopSizeDrag();
    updateBrushPreview();
  });

  stageWrap.addEventListener('touchstart', (e)=>{ if(e.touches.length>1) e.preventDefault(); }, {passive:false});

  // ---------- Panning ----------
  function startPan(e){
    previewSuppressedUntilMove = true;
    hideSizePreview();
    isPanning = true;
    panStart = {x: e.clientX, y: e.clientY, panX, panY};
    stageWrap.style.cursor = 'grab';
    stageWrap.setPointerCapture(e.pointerId);
  }
  function doPan(e){
    if(!panStart) return;
    panX = panStart.panX + (e.clientX - panStart.x);
    panY = panStart.panY + (e.clientY - panStart.y);
    applyTransform();
  }
  function stopPan(){
    previewSuppressedUntilMove = true;
    isPanning = false;
    panStart = null;
    updateIdleCursor();
    updateBrushPreview();
  }

  // ---------- Zoom-drag (Ctrl + Space + drag) ----------
  function startZoomDrag(e){
    isZoomDragging = true;
    const wrapRect = stageWrap.getBoundingClientRect();
    const anchorX = e.clientX - wrapRect.left;
    const anchorY = e.clientY - wrapRect.top;
    zoomDragStart = {
      startClientX: e.clientX, startZoom: zoom, anchorX, anchorY,
      canvasX: (anchorX - panX) / zoom, canvasY: (anchorY - panY) / zoom
    };
    stageWrap.style.cursor = 'ew-resize';
    stageWrap.setPointerCapture(e.pointerId);
  }
  function doZoomDrag(e){
    if(!zoomDragStart) return;
    // Zoom around the cursor's position from *before* the drag started —
    // not wherever the cursor is right now — so dragging left/right purely
    // controls zoom amount while the same point stays anchored under
    // where you began.
    const dx = e.clientX - zoomDragStart.startClientX;
    const factor = Math.pow(1.006, dx);
    let newZoom = zoomDragStart.startZoom * factor;
    newZoom = Math.min(Math.max(newZoom, 0.05), 30);
    panX = zoomDragStart.anchorX - zoomDragStart.canvasX * newZoom;
    panY = zoomDragStart.anchorY - zoomDragStart.canvasY * newZoom;
    zoom = newZoom;
    applyTransform();
  }
  function stopZoomDrag(){
    previewSuppressedUntilMove = true;
    isZoomDragging = false;
    zoomDragStart = null;
    updateIdleCursor();
    updateBrushPreview();
  }

  // ---------- Brush resize-drag (C + drag) ----------
  function startSizeDrag(e){
    isSizeDragging = true;
    const wrapRect = stageWrap.getBoundingClientRect();
    sizeDragStart = {
      clientX: e.clientX, clientY: e.clientY,
      size: getBrushSize(),
      anchorX: e.clientX - wrapRect.left,
      anchorY: e.clientY - wrapRect.top,
    };
    stageWrap.style.cursor = 'ew-resize';
    stageWrap.setPointerCapture(e.pointerId);
    showSizePreview(sizeDragStart.anchorX, sizeDragStart.anchorY, getBrushSize());
  }
  function doSizeDrag(e){
    if(!sizeDragStart) return;
    // Horizontal drag = coarse control, vertical = fine control — right and
    // down both grow the brush, left and up both shrink it.
    const dx = e.clientX - sizeDragStart.clientX;
    const dy = sizeDragStart.clientY - e.clientY;
    // Convert the screen-space drag into canvas-space size units. Without
    // this zoom correction resizing feels fast when zoomed in and slow when
    // zoomed out because one brush pixel occupies a different screen size.
    const delta = (dx * 0.6 - dy * 0.15) / Math.max(zoom, 0.01);
    let newSize = Math.round((sizeDragStart.size + delta) * 10) / 10;
    newSize = Math.min(Math.max(newSize, 0.1), 150);
    setBrushSize(newSize);
    sizeSlider.value = newSize;
    sizeVal.textContent = formatBrushSize(newSize);
    showSizePreview(sizeDragStart.anchorX, sizeDragStart.anchorY, getBrushSize());
  }
  function stopSizeDrag(){
    previewSuppressedUntilMove = true;
    isSizeDragging = false;
    sizeDragStart = null;
    updateIdleCursor();
    updateBrushPreview();
  }
  // Wrap-relative screen coords <-> backing-store canvas coords, without
  // the wrapRect subtraction screenToCanvas does (anchorX/anchorY here are
  // already wrap-relative).
  function wrapToCanvas(x, y){
    return { x: (x - panX) / zoom * SS, y: (y - panY) / zoom * SS };
  }
  function canvasToWrap(bx, by){
    return { x: (bx / SS) * zoom + panX, y: (by / SS) * zoom + panY };
  }

  function showSizePreview(anchorX, anchorY, size, pressure, fixedDiameter, outlineOnly = false){
    const PREVIEW_MIN_PRESSURE = 0.3;
    pressure = pressure == null
      ? 1
      : fixedDiameter == null
        ? Math.max(PREVIEW_MIN_PRESSURE, pressure)
        : Math.max(0, Math.min(1, pressure));
    // Render the preview at the SAME fixed backing-store density the real
    // canvas uses (size * SS, independent of zoom/DPR), then let CSS scale
    // it up by `zoom` — exactly how the real canvas is displayed. Using a
    // fresh dpr-scaled canvas every time (the old approach) always drew a
    // natively browser-smoothed circle, so it never matched what the actual
    // dab will look like once zoomed in past the "pixelated" threshold: the
    // real paint shows its true supersampled texels as visible blocks, but
    // the preview stayed silky smooth. Matching the backing resolution and
    // the image-rendering rule makes the preview a true, honest look-ahead.
    //
    // On top of that: the preview also needs to sit at the same PHASE on
    // the texel grid as a real dab would. Previously it was always drawn
    // dead-centered in its own little canvas and just followed the raw
    // mouse position — so its blocky pattern never actually shifted in sync
    // with the real canvas underneath as you moved the cursor, it just slid
    // around independently. Snapping the preview's origin to an integer
    // backing-store pixel (the same grid the real canvas rasterizes into)
    // and keeping the circle's sub-pixel offset within that box makes the
    // preview's antialiasing genuinely track the canvas as you move.
    //
    // The box stays sized for the full brush so its texel-grid phase remains
    // stable. The optional fixed diameter lets pen hover display a 1px
    // light-pressure dab without scaling up with the brush's maximum size.
    const margin = 4; // backing-store px of headroom for the AA edge
    const backingSize = Math.max(1, Math.round(size * SS)) + margin * 2;
    const rBacking = fixedDiameter == null
      ? pressureRadius(size, pressure)
      : Math.max(MIN_ABS_RADIUS_BACKING, fixedDiameter * SS / 2);

    const cursorCanvas = wrapToCanvas(anchorX, anchorY);
    const originBX = Math.floor(cursorCanvas.x - backingSize / 2);
    const originBY = Math.floor(cursorCanvas.y - backingSize / 2);
    const cBackingX = cursorCanvas.x - originBX; // fractional center within the box
    const cBackingY = cursorCanvas.y - originBY;

    const originWrap = canvasToWrap(originBX, originBY);
    const cssSize = backingSize / SS * zoom;

    sizePreview.style.display = 'block';
    sizePreview.style.left = originWrap.x + 'px';
    sizePreview.style.top = originWrap.y + 'px';
    sizePreview.style.width = cssSize + 'px';
    sizePreview.style.height = cssSize + 'px';
    // Same rule the main canvas uses: crisp blocky texels once zoomed in
    // past the threshold, smooth bilinear scaling below it.
    sizePreview.style.imageRendering = zoom >= 3.01 ? 'pixelated' : 'auto';

    if(sizePreview.width !== backingSize || sizePreview.height !== backingSize){
      sizePreview.width = backingSize;
      sizePreview.height = backingSize;
    }
    sizePreviewCtx.setTransform(1, 0, 0, 1, 0, 0);
    sizePreviewCtx.clearRect(0, 0, backingSize, backingSize);
    sizePreviewCtx.beginPath();
    sizePreviewCtx.arc(cBackingX, cBackingY, rBacking, 0, Math.PI * 2);

    if(outlineOnly){
      // Keep the outside edge at the exact pressure diameter. The ring is
      // only a live cursor indicator; it is never written into the stroke.
      const lineWidth = 1;
      sizePreviewCtx.lineWidth = lineWidth;
      sizePreviewCtx.strokeStyle = tool === 'eraser' ? '#ffffff' : color;
      sizePreviewCtx.beginPath();
      sizePreviewCtx.arc(cBackingX, cBackingY, Math.max(0, rBacking - lineWidth / 2), 0, Math.PI * 2);
      sizePreviewCtx.stroke();
    } else if(tool === 'eraser'){
      sizePreviewCtx.save();
      sizePreviewCtx.clip();
      sizePreviewCtx.fillStyle = '#fff';
      sizePreviewCtx.fillRect(0, 0, backingSize, backingSize);
      sizePreviewCtx.fillStyle = '#cfcfcf';
      const cell = 6 * SS;
      for(let y = 0; y < backingSize; y += cell){
        for(let x = 0; x < backingSize; x += cell){
          if(((x/cell + y/cell) % 2) === 0) sizePreviewCtx.fillRect(x, y, cell, cell);
        }
      }
      sizePreviewCtx.restore();
    } else {
      sizePreviewCtx.globalAlpha = pressureAlpha(pressure);
      sizePreviewCtx.fillStyle = color;
      sizePreviewCtx.fill();
    }
  }
  function hideSizePreview(){
    sizePreview.style.display = 'none';
  }

  // Live brush-size preview that tracks the pointer whenever it's hovering
  // the canvas (and isn't busy resizing/panning/zooming, which already show
  // their own preview via showSizePreview above).
  function updateBrushPreview(){
    if(isSizeDragging) return;
    if(isPanning){ hideSizePreview(); return; }
    if(drawing){
      const liveDiameter = 2 * pressureRadius(getBrushSize(), hoverPressure) / SS;
      showSizePreview(hoverX, hoverY, getBrushSize(), hoverPressure, liveDiameter, true);
      return;
    }
    if(isZoomDragging && zoomDragStart){
      // Pin the dab at the cursor position from *before* zooming started,
      // rather than tracking the live cursor while Ctrl+Space+drag is
      // actively changing the zoom.
      if(hoverPointerType === 'pen'){
        const lightestDiameter = 2 * MIN_ABS_RADIUS_BACKING / SS;
        showSizePreview(zoomDragStart.anchorX, zoomDragStart.anchorY, getBrushSize(), 0, lightestDiameter);
      } else {
        showSizePreview(zoomDragStart.anchorX, zoomDragStart.anchorY, getBrushSize());
      }
      return;
    }
    if(previewSuppressedUntilMove){ hideSizePreview(); return; }
    if(!hovering || hoverX === null) { hideSizePreview(); return; }
    if(hoverPointerType === 'pen'){
      // Match the brush engine's absolute lightest-pressure floor: a tiny
      // subpixel mark, independent of the selected maximum brush size.
      const lightestDiameter = 2 * MIN_ABS_RADIUS_BACKING / SS;
      showSizePreview(hoverX, hoverY, getBrushSize(), 0, lightestDiameter);
    } else {
      showSizePreview(hoverX, hoverY, getBrushSize());
    }
  }

  function isTypingTarget(el){
    if(!el) return false;
    const tag = el.tagName;
    return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || el.isContentEditable;
  }

  function updateIdleCursor(){
    if(isPanning || isZoomDragging || isSizeDragging) return;
    if(spaceDown && ctrlDown) stageWrap.style.cursor = 'zoom-in';
    else if(spaceDown) stageWrap.style.cursor = 'grab';
    else if(cDown) stageWrap.style.cursor = 'ew-resize';
    else stageWrap.style.cursor = 'none'; // the brush-size circle stands in for the cursor
  }

  window.addEventListener('keydown', (e)=>{
    if(isTypingTarget(e.target)) return;
    if(e.code === 'Space'){ spaceDown = true; e.preventDefault(); updateIdleCursor(); }
    if(e.key === 'Control'){ ctrlDown = true; updateIdleCursor(); }
    if(e.key.toLowerCase() === 'c' && !e.ctrlKey && !e.metaKey && !e.altKey){
      cDown = true; updateIdleCursor();
    }
    if(e.key.toLowerCase() === 'w' && !e.ctrlKey && !e.metaKey && !e.altKey){
      e.preventDefault();
      clearCanvas();
    }
    if(e.key.toLowerCase() === 'e' && !e.ctrlKey && !e.metaKey && !e.altKey){
      e.preventDefault();
      selectTool('eraser');
    }
    if(e.key.toLowerCase() === 'b' && !e.ctrlKey && !e.metaKey && !e.altKey){
      e.preventDefault();
      selectTool('pen');
    }
    if((e.ctrlKey || e.metaKey) && !e.shiftKey && e.key.toLowerCase()==='z'){
      e.preventDefault(); restore(undoStack, redoStack);
    }
    if((e.ctrlKey || e.metaKey) && (e.key.toLowerCase()==='y' || (e.shiftKey && e.key.toLowerCase()==='z'))){
      e.preventDefault(); restore(redoStack, undoStack);
    }
  });
  window.addEventListener('keyup', (e)=>{
    if(e.code === 'Space'){ spaceDown = false; updateIdleCursor(); }
    if(e.key === 'Control'){ ctrlDown = false; updateIdleCursor(); }
    if(e.key.toLowerCase() === 'c'){
      cDown = false;
      if(isSizeDragging) stopSizeDrag();
      updateIdleCursor();
    }
  });
  window.addEventListener('blur', ()=>{
    spaceDown=false; ctrlDown=false; cDown=false;
    if(isSizeDragging) stopSizeDrag();
    updateIdleCursor();
  });

  // ---------- Zoom (wheel) ----------
  stageWrap.addEventListener('wheel', (e)=>{
    e.preventDefault();
    const wrapRect = stageWrap.getBoundingClientRect();
    const mx = e.clientX - wrapRect.left;
    const my = e.clientY - wrapRect.top;
    const cx = (mx - panX) / zoom;
    const cy = (my - panY) / zoom;
    const delta = -e.deltaY;
    const factor = delta > 0 ? 1.08 : 1/1.08;
    let newZoom = zoom * factor;
    newZoom = Math.min(Math.max(newZoom, 0.05), 30);
    panX = mx - cx * newZoom;
    panY = my - cy * newZoom;
    zoom = newZoom;
    applyTransform();
    updateBrushPreview();
  }, {passive:false});

  // ---------- UI wiring ----------
  const toolPen = document.getElementById('toolPen');
  const toolEraser = document.getElementById('toolEraser');
  function selectTool(t){
    if(tool === t) return;
    tool = t;
    toolPen.classList.toggle('active', t === 'pen');
    toolEraser.classList.toggle('active', t === 'eraser');

  function zoomStep(factor){
    const wrapRect = stageWrap.getBoundingClientRect();
    // Anchor the zoom on the last known cursor position (if it's actually
    // within the canvas area) so the +/- buttons zoom the same way the
    // wheel does — staying under the cursor — instead of always zooming
    // around the screen's center.
    const mx = (hovering && hoverX !== null && hoverX >= 0 && hoverX <= wrapRect.width)
      ? hoverX : wrapRect.width/2;
    const my = (hovering && hoverY !== null && hoverY >= 0 && hoverY <= wrapRect.height)
      ? hoverY : wrapRect.height/2;
    const cx = (mx - panX)/zoom, cy = (my - panY)/zoom;
    zoom = Math.min(Math.max(zoom*factor, 0.05), 30);
    panX = mx - cx*zoom;
    panY = my - cy*zoom;
    applyTransform();
  }
